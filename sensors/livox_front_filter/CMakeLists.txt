cmake_minimum_required(VERSION 3.16)
project(livox_front_filter_local)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
add_executable(livox_front_filter main.cpp)
ament_target_dependencies(livox_front_filter rclcpp sensor_msgs livox_ros_driver2)
add_executable(test_front_filter test_front_filter.cpp)
ament_target_dependencies(test_front_filter sensor_msgs livox_ros_driver2)
enable_testing()
add_test(NAME front_geometry_and_metadata COMMAND test_front_filter)
