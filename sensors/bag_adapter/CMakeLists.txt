cmake_minimum_required(VERSION 3.8)
project(bag_adapter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
add_executable(pointcloud_to_livox src/pointcloud_to_livox.cpp)
target_compile_features(pointcloud_to_livox PUBLIC cxx_std_17)
target_include_directories(pointcloud_to_livox PRIVATE include)
ament_target_dependencies(pointcloud_to_livox rclcpp sensor_msgs livox_ros_driver2)
install(TARGETS pointcloud_to_livox DESTINATION lib/${PROJECT_NAME})
ament_package()
