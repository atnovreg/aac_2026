cmake_minimum_required(VERSION 3.16)
project(acados_trajectory_replay LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(TrajectoryReplay SHARED Replay.cc)
target_compile_features(TrajectoryReplay PRIVATE cxx_std_17)
target_link_libraries(TrajectoryReplay PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
add_executable(test_trajectory tests/test_trajectory.cpp)
target_include_directories(test_trajectory PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(test_trajectory PRIVATE cxx_std_17)
target_compile_options(test_trajectory PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME trajectory_interpolation COMMAND test_trajectory)
